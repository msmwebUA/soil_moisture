# copy lorawan-stack directory to server and cd into it

# clone default configs from github and copy to lorawan-stack directory
git clone https://github.com/chirpstack/chirpstack-docker /tmp/csd
cp -r /tmp/csd/configuration/chirpstack ./chirpstack/
cp -r /tmp/csd/configuration/chirpstack-gateway-bridge ./chirpstack-gateway-bridge/

# edit these rows in chirpstack.toml (use service names, not names of containers)
[postgresql]
dsn="postgres://chirpstack:chirpstack@postgres:5432/chirpstack?sslmode=disable"

[redis]
servers=["redis://redis:6379/"]

[integration.mqtt]
server="tcp://mosquitto:1883/"

# cd to dir contains docker-compose.yml file and run these commands:
sudo docker compose up -d postgres redis mosquitto        # wait minute to startup database
sudo docker compose logs -f postgres                      # you must see both the 'chirpstack' and 'thingsboard' databases get created (the init script runs once, only on a fresh/empty volume).
sudo docker compose up -d                                 # run other containers

-------------

# if chirpstack migration to postgres filed because of losed pg_trgm extension
# check extension exists
sudo docker run --rm   --network lorawan-stack_iot   -e PGPASSWORD=chirpstack   postgres:16-alpine   psql -h postgres -U chirpstack -d chirpstack   -c "SELECT extname, extversion FROM pg_extension WHERE extname = 'pg_trgm';"
# add extension to db
sudo docker run --rm   --network lorawan-stack_iot   -e PGPASSWORD=chirpstack   postgres:16-alpine   psql -h postgres -U chirpstack -d chirpstack   -c "CREATE EXTENSION IF NOT EXISTS pg_trgm;"

-------------
# if need to remove containers
sudo docker compose down

# if need to remove database volume
sudo docker volume ls                     # check volume name
sudo docker volume rm <volume_name>

# if need to remove all created columes
sudo docker compose down -v
